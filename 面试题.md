const Logger = (function () {
  let logMsg = "";
  console.log("x");
  return (msg?: any) => (msg ? (logMsg = `${logMsg}${msg}`) : logMsg);
})();

export default function App() {
  const [count, setCount] = useState(0);


  Logger("a");
  useEffect(() => {
    Logger("b");
    setInterval(() => {
      setCount((prev) => {
        if (prev === 0) return 0
        return prev
      });
    }, 1000);


  }, []);



  Logger("c");
  console.log(Logger());

  return <div>{count}</div>;
}
